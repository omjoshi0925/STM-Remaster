cmake_minimum_required(VERSION 3.24)
project(SpiderManTotalMayhemNative LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "16.0" CACHE STRING "")
set(CMAKE_OSX_ARCHITECTURES arm64)
set(CMAKE_XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1")
set(CMAKE_XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "local.totalmayhem.native")
set(CMAKE_XCODE_ATTRIBUTE_CODE_SIGN_STYLE "Automatic")
file(GLOB SRC CONFIGURE_DEPENDS Sources/*.mm Sources/*.m Sources/*.cpp Sources/*.c)
set(CMAKE_XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC "YES")
add_executable(SpiderManTotalMayhem MACOSX_BUNDLE ${SRC})
set_target_properties(SpiderManTotalMayhem PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in")
target_link_libraries(SpiderManTotalMayhem PRIVATE "-framework UIKit" "-framework Foundation" "-framework Metal" "-framework MetalKit" "-framework QuartzCore" "-framework CoreMotion" "-framework AVFoundation" "-framework AudioToolbox")
# Generated metadata is small and always bundled.
file(GLOB GENERATED_FILES "${CMAKE_CURRENT_SOURCE_DIR}/../Generated/*")
if(GENERATED_FILES)
  target_sources(SpiderManTotalMayhem PRIVATE ${GENERATED_FILES})
  set_source_files_properties(${GENERATED_FILES} PROPERTIES MACOSX_PACKAGE_LOCATION Resources/Generated)
endif()
# Prepared original assets are user-supplied and are staged at build time. Add them as a folder reference.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../Assets")
  file(GLOB_RECURSE GAME_RESOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../Assets/*")
  target_sources(SpiderManTotalMayhem PRIVATE ${GAME_RESOURCES})
  foreach(r ${GAME_RESOURCES})
    file(RELATIVE_PATH rel "${CMAKE_CURRENT_SOURCE_DIR}/../Assets" "${r}")
    get_filename_component(dir "${rel}" DIRECTORY)
    set_source_files_properties("${r}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/Assets/${dir}")
  endforeach()
endif()
file(GLOB NATIVE_RESOURCES "${CMAKE_CURRENT_SOURCE_DIR}/Resources/*")
if(NATIVE_RESOURCES)
  target_sources(SpiderManTotalMayhem PRIVATE ${NATIVE_RESOURCES})
  set_source_files_properties(${NATIVE_RESOURCES} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
